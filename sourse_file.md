# Программа для тренировки работы с файлами
# Задачи:
# - найти в текущей дирректории все файлы 'task_*' с расширением md
# - выделить из каждого файла блоки: "Номер задачи", "Задание" и "Решение"
# - подготовить блоки для записи в базу данных(временно пишем в файл)
# - сохранить историю в log файл

# подключаем модуль для работы с регулярными(шаблонными) строками
import re

# подключаем модуль работы с файлами и каталогами по шаблону
from glob import glob

# найти все файлы с расширением "md", начинающиеся на "task"
file_list = glob('task_*.md')
file_list.sort()

# компилируем регулярное выражение для блока "Задание"
task_reg = re.compile('Задача(.*)Решение', re.MULTILINE | re.DOTALL)
# компилируем регулярное выражение для блока "Решение"
decision_reg = re.compile('Решение(.*)Подсказки', re.MULTILINE | re.DOTALL)

for file_name in file_list:
    task_id = int(file_name.split('_')[1][:-3])  # получаем номер задачи из имени

    with open(file_name, 'r', encoding='utf-8') as f:
        file_text = f.read()  # сохраняем в строке все содержимое файла

    # Вырезаем по шаблону текст блока "Задание"
    task = re.findall(task_reg, file_text)[0].strip()
    # Вырезаем по шаблону текст блока "Решение"
    decision = re.findall(decision_reg, file_text)[0].strip()

    # Создаем файл и добавляем в него: "Номер задачи" "Задание" и "Решение"
    with open('all_tasks.md', 'a') as f:
        f.write(f'Задача №{task_id}\n\
    Задание {task_id}:\n{task}\n\
    Решение {task_id}:\n{decision}\n')

    # Добавляем в файл-log отчет об успешном выполнении итерации
    with open('tasks.log', 'a') as f:
        f.write(f'Задание из файла: {file_name} успешно загружено \n')
